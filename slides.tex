\section{历史沿革}
\begin{frame}{历史沿革}
  \begin{tikzpicture}
    \draw[thick,snnudarkred] (0,0) -- (9.5,0);
    \foreach \x/\up/\down in {
      0.25/陕西省立师范专科学校/1944,
      1.75/1954/更名西安师范学院,
      3.25/合并定名陕西师范大学/1960,
      4.75/1978/划归教育部直属,
      6.25/入选"211 工程"/2005,
      7.75/2017/首轮入选"双一流",
      9.25/再次入选"双一流"/2022}
    {
      \node[circle,fill=snnudarkred,inner sep=0pt,minimum size=2.5mm] at (\x,0) {};
      \node[above,align=center,text width=1.5cm,font=\scriptsize] at (\x,0.3) {\up};
      \node[below,align=center,text width=1.5cm,font=\scriptsize\bfseries] at (\x,-0.3) {\down};
    }
  \end{tikzpicture}
  \begin{itemize}
    \item 前身是 1944 年成立的陕西省立师范专科学校
    \item 1960 年,西安师范学院与陕西师范学院合并,定名为陕西师范大学
    \item 1978 年划归教育部直属,为六所部属师范大学之一
  \end{itemize}
\end{frame}
